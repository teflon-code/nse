1. LOCAL TRADINGVIEW WEBSOCKET / FASTAPI BRIDGE

TradingView limits browser cross-origin WebSocket & Scanner requests unless routed through a local bridge or CORS proxy. Run any of the open-source TradingView adapters below and connect this UI directly:

Cloud / Online API Endpoint (cloud_nse_api.py)
Local WebSocket Bridge (@mathieuc/tradingview)
Node.js Bridge (@mathieuc/tradingview)
npm i @mathieuc/tradingview ws
# server.js
const TradingView = require('@mathieuc/tradingview');
const WebSocket = require('ws');
const wss = new WebSocket.Server({ port: 8765 });
const client = new TradingView.Client();

wss.on('connection', (ws) => {
  ws.on('message', (raw) => {
    const { symbol = 'NSEKE:SCOM', timeframe = 'D', range = 1000 } = JSON.parse(raw);
    const chart = new client.Session.Chart();
    chart.setMarket(symbol, { timeframe, range });
    chart.onUpdate(() => {
      if (!chart.periods.length) return;
      const bars = chart.periods.slice().reverse().map(p => ({
        date: new Date(p.time * 1000).toISOString().slice(0, 16).replace('T', ' '),
        open: p.open, high: p.max, low: p.min, close: p.close, volume: p.volume
      }));
      ws.send(JSON.stringify({ type: 'history', symbol, timeframe, bars }));
    });
  });
});
Python Bridge (rongardF/tvdatafeed + FastAPI)
pip install fastapi uvicorn git+https://github.com/rongardF/tvdatafeed.git
# api.py
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from tvDatafeed import TvDatafeed, Interval

app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])
tv = TvDatafeed()

@app.get("/history")
def history(symbol: str = "SCOM", interval: str = "1D", bars: int = 1000):
    df = tv.get_hist(symbol=symbol, exchange="NSEKE", interval=Interval.in_daily, n_bars=bars)
    return [{"date": str(idx)[:16], "open": r.open, "high": r.high, "low": r.low, "close": r.close, "volume": r.volume} for idx, r in df.iterrows()]
